cmake_minimum_required(VERSION 3.8)
project(boundary_sweep_workspace_analysis)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
add_executable(boundary_sweep_workspace src/boundary_sweep_workspace.cpp)
ament_target_dependencies(boundary_sweep_workspace rclcpp moveit_core moveit_ros_planning)
install(TARGETS boundary_sweep_workspace DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/postprocess_boundary_sweep_workspace.py scripts/boundary_sweep_workspace_rviz.py scripts/generate_boundary_sweep_workspace_video.py scripts/finalize_boundary_sweep_workspace.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch rviz DESTINATION share/${PROJECT_NAME})
ament_package()
